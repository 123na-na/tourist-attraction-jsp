-- 创建数据库
CREATE DATABASE IF NOT EXISTS `tourism_system` CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci;
USE `tourism_system`;

SET NAMES utf8mb4;
SET FOREIGN_KEY_CHECKS = 0;

-- 先删除所有表（按依赖顺序）
DROP TABLE IF EXISTS `shopping_cart`;
DROP TABLE IF EXISTS `tb_goods`;
DROP TABLE IF EXISTS `goods`;
DROP TABLE IF EXISTS `tour_group`;
DROP TABLE IF EXISTS `scenic_spot`;
DROP TABLE IF EXISTS `user`;
DROP TABLE IF EXISTS `tb_type`;
DROP TABLE IF EXISTS `category`;

-- 创建 category 表
CREATE TABLE `category` (
  `category_id` int NOT NULL AUTO_INCREMENT,
  `category_name` varchar(50) NOT NULL,
  PRIMARY KEY (`category_id`),
  UNIQUE KEY `category_name` (`category_name`)
) ENGINE=InnoDB AUTO_INCREMENT=1 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;

-- 创建 scenic_spot 表
CREATE TABLE `scenic_spot` (
  `spot_id` int NOT NULL AUTO_INCREMENT,
  `spot_name` varchar(100) NOT NULL,
  `location_province` varchar(50) NOT NULL,
  `location_city` varchar(50) NOT NULL,
  `description` text,
  `ticket_price` decimal(8,2) NOT NULL,
  `opening_time` varchar(50) NOT NULL,
  `image_url` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`spot_id`)
) ENGINE=InnoDB AUTO_INCREMENT=1 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;

-- 创建 tb_goods 表
CREATE TABLE `tb_goods` (
  `goodsId` int NOT NULL AUTO_INCREMENT,
  `goodsName` varchar(100) NOT NULL,
  `start_date` date NOT NULL,
  `end_date` date NOT NULL,
  `price` decimal(10,2) NOT NULL,
  `guide` varchar(50) DEFAULT NULL,
  `status` enum('计划中','进行中','已结束') NOT NULL DEFAULT '计划中',
  `category_id` int DEFAULT NULL,
  `location` varchar(255) NOT NULL,
  `image` varchar(45) DEFAULT NULL,
  PRIMARY KEY (`goodsId`),
  KEY `category_id` (`category_id`),
  CONSTRAINT `fk_tb_goods_category` FOREIGN KEY (`category_id`) REFERENCES `category` (`category_id`)
) ENGINE=InnoDB AUTO_INCREMENT=1 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;

-- 创建 user 表
CREATE TABLE `user` (
  `userid` int NOT NULL AUTO_INCREMENT,
  `username` varchar(50) NOT NULL,
  `password` varchar(32) NOT NULL,
  `email` varchar(100) DEFAULT NULL,
  `phone` varchar(20) DEFAULT NULL,
  `role` enum('user','admin','group') NOT NULL DEFAULT 'user',
  `create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`userid`),
  UNIQUE KEY `username` (`username`),
  UNIQUE KEY `email` (`email`),
  UNIQUE KEY `phone` (`phone`)
) ENGINE=InnoDB AUTO_INCREMENT=1 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;

-- 创建 tb_type 表
CREATE TABLE `tb_type` (
  `typeId` int NOT NULL AUTO_INCREMENT,
  `typeName` varchar(45) NOT NULL,
  PRIMARY KEY (`typeId`),
  UNIQUE KEY `typeName_UNIQUE` (`typeName`)
) ENGINE=InnoDB AUTO_INCREMENT=56 DEFAULT CHARSET=utf8mb3;

-- 创建 shopping_cart 表（已包含用户姓名和商品姓名）
CREATE TABLE `shopping_cart` (
  `cart_item_id` int NOT NULL AUTO_INCREMENT,
  `userid` int NOT NULL COMMENT '用户ID',
  `user_name` varchar(50) NOT NULL COMMENT '用户姓名（冗余字段，避免关联查询）',
  `goodsId` int NOT NULL COMMENT '商品ID',
  `goods_name` varchar(100) NOT NULL COMMENT '商品名称（冗余字段，避免关联查询）',
  `goods_price` decimal(10,2) NOT NULL COMMENT '商品单价（冗余字段，避免关联查询）',
  `quantity` int NOT NULL DEFAULT 1 COMMENT '商品数量',
  `selected_date` date DEFAULT NULL COMMENT '选择的旅游日期',
  `travelers_count` int DEFAULT 1 COMMENT '出行人数',
  `add_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '加入购物车时间',
  `update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (`cart_item_id`),
  UNIQUE KEY `unique_user_goods` (`userid`, `goodsId`) COMMENT '防止同一用户重复添加同一商品',
  KEY `idx_user_id` (`userid`),
  KEY `fk_cart_goods` (`goodsId`),
  CONSTRAINT `fk_cart_user` FOREIGN KEY (`userid`) REFERENCES `user` (`userid`) ON DELETE CASCADE,
  CONSTRAINT `fk_cart_goods` FOREIGN KEY (`goodsId`) REFERENCES `tb_goods` (`goodsId`) ON DELETE CASCADE,
  CONSTRAINT `chk_quantity` CHECK (`quantity` > 0),
  CONSTRAINT `chk_travelers_count` CHECK (`travelers_count` > 0)
) ENGINE=InnoDB AUTO_INCREMENT=1 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='购物车表';
